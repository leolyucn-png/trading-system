### 🐍 Krippendorff's α 计算脚本（我写好了，你填完数据跑）

```python
# krippendorff_alpha.py
# 用法：填入数据后运行，输出α系数
# 需求：pip install krippendorff pandas openpyxl

import pandas as pd
import krippendorff

# 读取填好的工作簿
df = pd.read_excel("投资者情感_人工编码工作簿.xlsx", sheet_name="Coding")

# 只取E列和F列都有值的行
mask = df["Coder A"].notna() & df["Coder B"].notna()
coded = df[mask]

# 编码映射（nominal -> numeric）
code_map = {
    "Information sharing": 1,
    "Policy evaluation": 2,
    "Market expectation": 3,
    "Emotional venting": 4,
    "Other/Mixed": 5
}

coder_a = coded["Coder A"].map(code_map).values
coder_b = coded["Coder B"].map(code_map).values

# Krippendorff's α (nominal)
alpha = krippendorff.alpha(
    reliability_values=[coder_a, coder_b],
    level_of_measurement="nominal"
)

n_total = len(coded)
n_agree = (coder_a == coder_b).sum()
pct_agree = n_agree / n_total

print(f"双编码总行数: {n_total}")
print(f"一致行数: {n_agree}")
print(f"简单一致率: {pct_agree:.2%}")
print(f"Krippendorff's α: {alpha:.3f}")
print(f"目标 α ≥ 0.80 → {'✅ PASS' if alpha >= 0.80 else '❌ NEED REVISION' if alpha < 0.667 else '⚠️ ACCEPTABLE (exploratory)'}")

# 输出最终分布
final = coded["Coder A"].copy()
# 裁决优先：如果G列有值，用G列
g_mask = df["Final (adjudicated)"].notna()
final.loc[g_mask] = df.loc[g_mask, "Final (adjudicated)"]
print(f"\n最终分布:")
print(final.value_counts())
```

### 🐍 SnowNLP复现脚本（你跑在你的数据上）

```python
# snownlp_replicate.py
# 用法：指定评论列表文件，输出情感分布+Fig 5数据+阈值文档
# 需求：pip install snownlp pandas matplotlib

import pandas as pd
import numpy as np
from snownlp import SnowNLP
import matplotlib.pyplot as plt
import json

# === 配置 ===
INPUT_FILE = "comments_85770.csv"  # 你的85,770条数据
POS_THRESHOLD = 0.6    # 正性阈值 ← 你调这里
NEG_THRESHOLD = 0.4    # 负性阈值 ← 你调这里
STAGE_BOUNDARIES = {
    "Stage1": ("2024-09-01", "2024-09-23"),
    "Stage2": ("2024-09-24", "2024-10-27"),
    "Stage3": ("2024-10-28", "2024-12-16"),
    "Stage4": ("2024-12-17", "2025-01-28"),
}

# === 1. 读取数据 ===
df = pd.read_csv(INPUT_FILE)
print(f"总评论数: {len(df)}")

# === 2. 情感打分 ===
def get_sentiment(text):
    try:
        return SnowNLP(str(text)).sentiments
    except:
        return np.nan

# 如果数据量大，加tqdm进度条
# from tqdm import tqdm
# tqdm.pandas()
# df["snow_score"] = df["comment"].progress_apply(get_sentiment)
df["snow_score"] = df["comment"].apply(get_sentiment)
valid = df["snow_score"].notna()
print(f"有效打分: {valid.sum()}/{len(df)}")
print(f"均值: {df.loc[valid, 'snow_score'].mean():.3f}")
print(f"中位数: {df.loc[valid, 'snow_score'].median():.3f}")
print(f"标准差: {df.loc[valid, 'snow_score'].std():.3f}")

# === 3. 阈值分类 ===
def classify(score):
    if score > POS_THRESHOLD:
        return "positive"
    elif score < NEG_THRESHOLD:
        return "negative"
    else:
        return "neutral"

df["polarity"] = df["snow_score"].apply(classify)

# === 4. 分阶段统计 ===
df["date"] = pd.to_datetime(df["date"])
results = []
for stage, (start, end) in STAGE_BOUNDARIES.items():
    mask = (df["date"] >= start) & (df["date"] <= end)
    subset = df[mask]
    counts = subset["polarity"].value_counts()
    total = len(subset)
    results.append({
        "stage": stage,
        "n": total,
        "positive_pct": round(counts.get("positive", 0) / total * 100, 1),
        "negative_pct": round(counts.get("negative", 0) / total * 100, 1),
        "neutral_pct": round(counts.get("neutral", 0) / total * 100, 1),
        "mean_score": round(subset["snow_score"].mean(), 3),
        "score_std": round(subset["snow_score"].std(), 3),
    })

result_df = pd.DataFrame(results)
print("\n=== 分阶段情感分布 ===")
print(result_df.to_string(index=False))

# === 5. 极化指标 ===
# 情感极化 = 1 - (positive_pct - negative_pct)的绝对值 / (positive_pct + negative_pct)
# 值越高=越极化
result_df["polarization"] = 1 - abs(result_df["positive_pct"] - result_df["negative_pct"]) / (result_df["positive_pct"] + result_df["negative_pct"])

# === 6. 输出阈值文档（改稿子用） ===
threshold_doc = f"""
## SnowNLP 情感阈值声明

### 分类标准
正性 (Positive): SnowNLP score > {POS_THRESHOLD}
负性 (Negative): SnowNLP score < {NEG_THRESHOLD}
中性 (Neutral): {NEG_THRESHOLD} ≤ score ≤ {POS_THRESHOLD}

### 有效性验证
总有效打分: {valid.sum()}
情感得分均值: {df.loc[valid, 'snow_score'].mean():.3f}
标准差: {df.loc[valid, 'snow_score'].std():.3f}

### 各阶段指标
{result_df.to_string(index=False)}
"""

with open("snownlp_thresholds_report.txt", "w") as f:
    f.write(threshold_doc)
print("\n✅ 阈值报告已保存: snownlp_thresholds_report.txt")

# === 7. 可选：人工标注验证子集F1 ===
# 如果还有时间，手工标200条做F1验证
# 脚本会生成200条随机抽样，你标完跑这个：
def f1_validation():
    """人工标注验证 — 标完200条后跑这个"""
    gold = pd.read_excel("snownlp_validation_200.xlsx")  # 你标的
    pred = df[df["id"].isin(gold["id"])]
    
    from sklearn.metrics import classification_report, f1_score
    print(classification_report(gold["gold_label"], pred["polarity"]))
    print(f"Macro F1: {f1_score(gold['gold_label'], pred['polarity'], average='macro'):.3f}")
```
